# ZXMetalCircle - configuración del firmware de la Raspberry Pi
# https://www.raspberrypi.com/documentation/computers/config_txt.html
#
# Cada modelo arranca su kernel: la Pi 1, Zero y Pi 2 en 32 bits; la Pi 3,
# Zero 2 W, 4, 400, 5 y 500 en 64 bits.
#
# Para volver a 32 bits en la Pi 3 / Zero 2 W o en la Pi 4 / 400, cambia en su
# sección arm_64bit=1 por arm_64bit=0 y quita las líneas kernel_address,
# kernel y armstub de 64 bits (las alternativas de 32 bits están comentadas).
# La Pi 5 y 500 solo funcionan en 64 bits.

arm_64bit=0
initial_turbo=0
disable_overscan=1

# Arranque: sin la pantalla de colores del firmware y sin su espera antes de
# cargar el emulador
disable_splash=1
boot_delay=0

# Filtro de escalado de la GPU: 8 = vecino más próximo, píxeles nítidos. Sin
# esta línea la GPU suaviza la imagen al ampliarla y las letras se ven borrosas.
#  1 sinc
#  2 sinc Blackman
#  3 sinc sin lóbulos laterales
#  4 sinc con medio primer lóbulo lateral
#  5 sinc Hamming
#  6 sinc Hamming 3pi (el de la GPU si no se pone nada)
#  7 sinc Hamming 2,5pi
#  8 vecino más próximo
scaling_kernel=8

enable_uart=1

# Pi 3 y Zero 2 W: 64 bits (32 bits: arm_64bit=0 y kernel=kernel8-32.img)
[pi3]
arm_64bit=1
kernel_address=0x80000
kernel=kernel8.img

[pi02]
arm_64bit=1
kernel_address=0x80000
kernel=kernel8.img

# Pi 4 y 400: 64 bits (32 bits: arm_64bit=0, kernel=kernel7l.img y
# armstub=armstub7-rpi4.bin)
[pi4]

kernel_address=0x80000
kernel=kernel8-rpi4.img
armstub=armstub8-rpi4.bin
max_framebuffers=2
